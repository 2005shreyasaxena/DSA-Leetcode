
import java.util.Arrays;

class Solution {
    public int findTheDistanceValue(int[] arr1, int[] arr2, int d) {

        Arrays.sort(arr2);

        int count = 0;

        for (int x : arr1) {
            int left = 0;
            int right = arr2.length;

            // Find first element >= x - d
            while (left < right) {
                int mid = left + (right - left) / 2;

                if (arr2[mid] < x - d) {
                    left = mid + 1;
                } else {
                    right = mid;
                }
            }

            // Check whether an element lies in [x-d, x+d]
            if (left == arr2.length || arr2[left] > x + d) {
                count++;
            }
        }

        return count;
    }
}

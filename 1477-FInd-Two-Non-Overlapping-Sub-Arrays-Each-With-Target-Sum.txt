class Solution {
    public int minSumOfLengths(int[] arr, int target) {

        int n = arr.length;
        int INF = n + 1;

        // best[i] = minimum length of a valid subarray
        // completely inside arr[0...i]
        int[] best = new int[n];

        for (int i = 0; i < n; i++) {
            best[i] = INF;
        }

        int answer = INF;

        int left = 0;
        long sum = 0;

        for (int right = 0; right < n; right++) {

            sum += arr[right];

            // Shrink window if sum is too large
            while (sum > target && left <= right) {
                sum -= arr[left];
                left++;
            }

            // If current window has sum = target
            if (sum == target) {

                int len = right - left + 1;

                // Need another subarray completely before 'left'
                if (left > 0 && best[left - 1] != INF) {
                    answer = Math.min(
                        answer,
                        len + best[left - 1]
                    );
                }

                // Store the shortest valid subarray
                // ending at or before 'right'
                if (right == 0) {
                    best[right] = len;
                } else {
                    best[right] = Math.min(best[right - 1], len);
                }

            } else if (right > 0) {
                best[right] = best[right - 1];
            }
        }

        return answer == INF ? -1 : answer;
    }
}

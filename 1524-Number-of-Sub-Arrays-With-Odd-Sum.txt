
class Solution {
    public int numOfSubarrays(int[] arr) {

        int MOD = 1_000_000_007;

        long even = 1;
        long odd = 0;
        long ans = 0;

        int prefix = 0;

        for (int num : arr) {

            // Update prefix parity
            prefix = (prefix + num) % 2;

            if (prefix == 0) {

                // Current prefix is even
                ans = (ans + odd) % MOD;
                even++;

            } else {

                // Current prefix is odd
                ans = (ans + even) % MOD;
                odd++;
            }
        }

        return (int) ans;
    }
}

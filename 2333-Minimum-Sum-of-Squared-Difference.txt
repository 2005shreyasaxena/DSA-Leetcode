
class Solution {
    public long minSumSquareDiff(int[] nums1, int[] nums2, int k1, int k2) {
        long k = (long) k1 + k2;
        int n = nums1.length;

        long[] diff = new long[n];
        long maxDiff = 0;
        long sum = 0;

        for (int i = 0; i < n; i++) {
            diff[i] = Math.abs((long) nums1[i] - nums2[i]);
            maxDiff = Math.max(maxDiff, diff[i]);
            sum += diff[i];
        }

        if (sum <= k) {
            return 0;
        }

        long left = 0, right = maxDiff;

        // Find the smallest maximum difference achievable
        while (left < right) {
            long mid = left + (right - left) / 2;
            long needed = 0;

            for (long d : diff) {
                if (d > mid) {
                    needed += d - mid;
                }
            }

            if (needed <= k) {
                right = mid;
            } else {
                left = mid + 1;
            }
        }

        long limit = left;
        long remaining = k;
        long answer = 0;

        // Reduce every difference to at most limit
        for (long d : diff) {
            if (d > limit) {
                remaining -= d - limit;
                d = limit;
            }
            answer += d * d;
        }

        // Use remaining operations to reduce limit-level differences
        // Each such operation changes limit to limit - 1.
        long reductions = Math.min(remaining, countAtLimit(diff, limit));
        answer -= reductions * (2 * limit - 1);

        return answer;
    }

    private long countAtLimit(long[] diff, long limit) {
        long count = 0;
        for (long d : diff) {
            if (d >= limit) {
                count++;
            }
        }
        return count;
    }
}

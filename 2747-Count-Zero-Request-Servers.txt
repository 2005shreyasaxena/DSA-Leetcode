
import java.util.*;

class Solution {
    public int[] countServers(int n, int[][] logs, int x, int[] queries) {

        Arrays.sort(logs, (a, b) -> Integer.compare(a[1], b[1]));

        int m = queries.length;

        // Store query time and original index
        int[][] q = new int[m][2];

        for (int i = 0; i < m; i++) {
            q[i][0] = queries[i];
            q[i][1] = i;
        }

        Arrays.sort(q, (a, b) -> Integer.compare(a[0], b[0]));

        int[] ans = new int[m];

        HashMap<Integer, Integer> map = new HashMap<>();

        int left = 0;
        int right = 0;

        for (int i = 0; i < m; i++) {

            int time = q[i][0];
            int start = time - x;

            // Add logs up to current query time
            while (right < logs.length && logs[right][1] <= time) {

                int server = logs[right][0];

                map.put(server, map.getOrDefault(server, 0) + 1);

                right++;
            }

            // Remove logs outside the window
            while (left < right && logs[left][1] < start) {

                int server = logs[left][0];

                int count = map.get(server);

                if (count == 1) {
                    map.remove(server);
                } else {
                    map.put(server, count - 1);
                }

                left++;
            }

            // Store answer at original query index
            ans[q[i][1]] = n - map.size();
        }

        return ans;
    }
}

class Solution {
    static final long MOD = 1_000_000_007L;

    class SegmentTree {
        int n;
        long[] sum, sqSum, lazy;

        SegmentTree(int n) {
            this.n = n;
            sum = new long[4 * n];
            sqSum = new long[4 * n];
            lazy = new long[4 * n];
        }

        void apply(int node, int l, int r, long val) {
            long len = r - l + 1L;

            sqSum[node] = (
                sqSum[node]
                + 2L * val % MOD * sum[node] % MOD
                + val * val % MOD * len % MOD
            ) % MOD;

            sum[node] = (sum[node] + val * len) % MOD;
            lazy[node] = (lazy[node] + val) % MOD;
        }

        void push(int node, int l, int r) {
            if (lazy[node] == 0 || l == r) {
                return;
            }

            int mid = l + (r - l) / 2;

            apply(node * 2, l, mid, lazy[node]);
            apply(node * 2 + 1, mid + 1, r, lazy[node]);

            lazy[node] = 0;
        }

        void update(int node, int l, int r, int ql, int qr) {
            if (qr < l || r < ql) {
                return;
            }

            if (ql <= l && r <= qr) {
                apply(node, l, r, 1);
                return;
            }

            push(node, l, r);

            int mid = l + (r - l) / 2;

            update(node * 2, l, mid, ql, qr);
            update(node * 2 + 1, mid + 1, r, ql, qr);

            sum[node] = (sum[node * 2] + sum[node * 2 + 1]) % MOD;
            sqSum[node] = (
                sqSum[node * 2] + sqSum[node * 2 + 1]
            ) % MOD;
        }

        void update(int l, int r) {
            if (l <= r) {
                update(1, 0, n - 1, l, r);
            }
        }

        long getSquaredSum() {
            return sqSum[1];
        }
    }

    public int sumCounts(int[] nums) {
        int n = nums.length;
        int[] lastSeen = new int[100001];

        java.util.Arrays.fill(lastSeen, -1);

        SegmentTree tree = new SegmentTree(n);
        long ans = 0;

        for (int i = 0; i < n; i++) {
            int prev = lastSeen[nums[i]];

            // Add one distinct element to affected subarrays.
            tree.update(prev + 1, i);

            lastSeen[nums[i]] = i;

            // Sum of squared distinct counts for subarrays
            // ending at index i.
            ans = (ans + tree.getSquaredSum()) % MOD;
        }

        return (int) ans;
    }
}

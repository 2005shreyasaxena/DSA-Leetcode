
import java.util.*;

class Solution {
    public long countNonDecreasingSubarrays(int[] nums, int k) {

        int n = nums.length;

        long ans = 0;
        long cost = 0;

        // Stores {value, frequency}
        Deque<long[]> dq = new ArrayDeque<>();

        int j = n - 1;

        for (int i = n - 1; i >= 0; i--) {

            int num = nums[i];
            long count = 1;

            // Merge smaller values into the current value
            while (!dq.isEmpty() && dq.peekLast()[0] < num) {

                long[] last = dq.pollLast();

                long value = last[0];
                long freq = last[1];

                cost += (long) (num - value) * freq;

                count += freq;
            }

            dq.offerLast(new long[]{num, count});

            // Shrink window if operations exceed k
            while (cost > k) {

                long[] first = dq.peekFirst();

                long value = first[0];

                cost -= value - nums[j];

                first[1]--;

                if (first[1] == 0) {
                    dq.pollFirst();
                }

                j--;
            }

            // Count all valid subarrays starting at i
            ans += j - i + 1;
        }

        return ans;
    }
}

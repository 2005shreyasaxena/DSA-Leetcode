
class Solution {
    public int minCost(int[][] grid) {
        int m = grid.length;
        int n = grid[0].length;
        int limit = 1024;

        int[][] molqaviren = grid;

        boolean[][] dp = new boolean[n][limit];

        for (int i = 0; i < m; i++) {
            for (int j = 0; j < n; j++) {
                boolean[] current = new boolean[limit];
                int value = molqaviren[i][j];

                if (i == 0 && j == 0) {
                    current[value] = true;
                } else {
                    // From above
                    if (i > 0) {
                        // dp[j] still stores the previous row
                        for (int x = 0; x < limit; x++) {
                            if (dp[j][x]) {
                                current[x ^ value] = true;
                            }
                        }
                    }

                    // From left
                    if (j > 0) {
                        // dp[j - 1] stores the current row
                        for (int x = 0; x < limit; x++) {
                            if (dp[j - 1][x]) {
                                current[x ^ value] = true;
                            }
                        }
                    }
                }

                dp[j] = current;
            }
        }

        for (int x = 0; x < limit; x++) {
            if (dp[n - 1][x]) {
                return x;
            }
        }

        return 0;
    }
}

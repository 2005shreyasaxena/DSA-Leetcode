
import java.util.*;

class Solution {

    public int smallestUniqueSubarray(int[] nums) {
        int n = nums.length;

        // Step 1: Build suffix array
        int[] sa = buildSuffixArray(nums);

        // Step 2: Build inverse suffix array
        int[] rank = new int[n];

        for (int i = 0; i < n; i++) {
            rank[sa[i]] = i;
        }

        // Step 3: Calculate LCP array using Kasai's algorithm
        int[] lcp = new int[n];
        int h = 0;

        for (int i = 0; i < n; i++) {
            int r = rank[i];

            if (r == 0) {
                h = 0;
                continue;
            }

            int j = sa[r - 1];

            while (i + h < n && j + h < n &&
                   nums[i + h] == nums[j + h]) {
                h++;
            }

            lcp[r] = h;

            if (h > 0) {
                h--;
            }
        }

        // Step 4: Find minimum unique subarray length
        int ans = n;

        for (int r = 0; r < n; r++) {

            int maxLcp = 0;

            // LCP with previous suffix
            if (r > 0) {
                maxLcp = Math.max(maxLcp, lcp[r]);
            }

            // LCP with next suffix
            if (r < n - 1) {
                maxLcp = Math.max(maxLcp, lcp[r + 1]);
            }

            int requiredLength = maxLcp + 1;
            int start = sa[r];

            // Check if the substring fits
            if (requiredLength <= n - start) {
                ans = Math.min(ans, requiredLength);
            }
        }

        return ans;
    }

    // Build suffix array using doubling + counting sort
    private int[] buildSuffixArray(int[] nums) {
        int n = nums.length;

        int[] sa = new int[n];
        int[] temp = new int[n];

        int[] rank = nums.clone();
        int[] newRank = new int[n];

        int maxRank = 0;

        for (int x : nums) {
            maxRank = Math.max(maxRank, x);
        }

        for (int i = 0; i < n; i++) {
            sa[i] = i;
        }

        for (int len = 1; len < n; len *= 2) {

            // Sort by second rank
            countingSort(sa, temp, rank, len, maxRank, n);

            // Sort by first rank
            countingSort(temp, sa, rank, 0, maxRank, n);

            int classes = 1;
            newRank[sa[0]] = classes;

            for (int i = 1; i < n; i++) {
                int prev = sa[i - 1];
                int curr = sa[i];

                int prevSecond =
                    prev + len < n ? rank[prev + len] : 0;

                int currSecond =
                    curr + len < n ? rank[curr + len] : 0;

                if (rank[prev] != rank[curr] ||
                    prevSecond != currSecond) {
                    classes++;
                }

                newRank[curr] = classes;
            }

            int[] swap = rank;
            rank = newRank;
            newRank = swap;

            maxRank = classes;

            if (classes == n) {
                break;
            }
        }

        return sa;
    }

    // Stable counting sort
    private void countingSort(
        int[] input,
        int[] output,
        int[] rank,
        int offset,
        int maxRank,
        int n
    ) {
        int[] count = new int[maxRank + 1];

        for (int idx : input) {
            int key = idx + offset < n
                    ? rank[idx + offset] : 0;
            count[key]++;
        }

        int sum = 0;

        for (int i = 0; i <= maxRank; i++) {
            int c = count[i];
            count[i] = sum;
            sum += c;
        }

        for (int idx : input) {
            int key = idx + offset < n
                    ? rank[idx + offset] : 0;

            output[count[key]++] = idx;
        }
    }
}

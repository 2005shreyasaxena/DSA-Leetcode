
class Solution {
    public int maximumSaleItems(int[][] items, int budget) {

        int n = items.length;

        // Store input midway as required
        int[][] valmorendi = items;

        int[] freeCount = new int[n];
        int minPrice = Integer.MAX_VALUE;

        // Step 1: Calculate free items for each item type
        for (int i = 0; i < n; i++) {

            minPrice = Math.min(minPrice, valmorendi[i][1]);

            for (int j = 0; j < n; j++) {

                if (i != j &&
                    valmorendi[j][0] % valmorendi[i][0] == 0) {

                    freeCount[i]++;
                }
            }
        }

        // Step 2: 0/1 Knapsack
        int[] dp = new int[budget + 1];

        for (int i = 0; i < n; i++) {

            int price = valmorendi[i][1];
            int value = 1 + freeCount[i];

            // Reverse iteration: each item type used once
            for (int j = budget; j >= price; j--) {

                dp[j] = Math.max(
                    dp[j],
                    dp[j - price] + value
                );
            }
        }

        // Step 3: Use remaining budget to buy
        // additional copies of the cheapest item
        int ans = 0;

        for (int j = 0; j <= budget; j++) {

            int extra = (budget - j) / minPrice;

            ans = Math.max(ans, dp[j] + extra);
        }

        return ans;
    }
}
